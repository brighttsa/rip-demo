Calibrating a sensor

Apply the stored offset and scale before you publish a reading.

def apply_calibration(raw, offset=0.42, scale=1.003):
    if raw is None:
        return None
    return (raw - offset) * scale


reading = apply_calibration(21.9)
print(round(reading, 2))
